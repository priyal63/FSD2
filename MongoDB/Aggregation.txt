Question 1
-- Calculate the average order amount for each category with amount greater than 10000. Sort the result in descending order of average amount and display only category and average amount.
--> db.orders.aggregate([
{
$match:{amount:{$gt:10000}}
},
{
   $group:{
      _id:"$category",
      averageAmount:{$avg:"$amount"}
   }
},
{
   $sort:{averageAmount:-1}
},
{
   $project:{
      category:"$_id",
      AverageAmount:"$averageAmount",
      _id:0
   }
}
])

===========================================================================================================================

Question 2
-- Calculate the total sales amount for each category. Sort the result in descending order of total sales and display only category and total sales.
Solution
--> db.orders.aggregate([
{
   $group:{
      _id:"$category",
      totalSales:{$sum:"$amount"}
   }
},
{
   $sort:{totalSales:-1}
},
{
   $project:{
      Category:"$_id",
      TotalSales:"$totalSales",
      _id:0
   }
}
])

===========================================================================================================================

Question 3
-- Find the highest order amount for each category. Sort the result in descending order of highest order amount and display only the category and highest order amount.
Solution
--> db.orders.aggregate([
{
   $group:{
      _id:"$category",
      highestAmount:{
         $max:"$amount"
      }
   }
},
{
   $sort:{
      highestAmount:-1
   }
},
{
   $project:{
      Category:"$_id",
      HighestAmount:"$highestAmount",
      _id:0
   }
}
])
